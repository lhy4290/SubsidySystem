53 private EligibilityResponse evaluate(EligibilityRequest req, boolean record) {
54 // 單次取時，所有檢查共用；統一轉換為 Asia/Taipei，不依賴 Clock 時區
55 ZonedDateTime now = ZonedDateTime.now(clock).withZoneSameInstant(TAIPEI);
56 LocalDate today = now.toLocalDate();
57
58 Optional<SubsidyType> type = SubsidyType.parse(req.subsidyType());
59 if (type.isEmpty()) return result(INVALID_SUBSIDY_TYPE, now, null);
60
61 Optional<LocalDate> birthDate = parseBirthDate(req.birthDate(), today);
62 if (birthDate.isEmpty()) return result(INVALID_BIRTH_DATE, now, null);
63
64 String nationalId = NationalIdValidator.normalize(req.nationalId());
65 if (!NationalIdValidator.isValid(nationalId)) return result(INVALID_NATIONAL_ID, now, null);
66
67 // Period 計算至日：生日當日滿歲；2/29 出生者於平年以 3/1 起算
68 if (Period.between(birthDate.get(), today).getYears() < MIN_AGE) return result(UNDER_AGE, now, null);
69
70 LocalTime time = now.toLocalTime();
71 if (time.isBefore(OPEN) || !time.isBefore(CLOSE))
72 return result(OUT_OF_SERVICE_HOURS, now, nextAvailableAt(type.get(), now));
73
74 if (!type.get().isOpenOn(today.getDayOfMonth()))
75 return result(OUT_OF_APPLICATION_PERIOD, now, nextAvailableAt(type.get(), now));
76
77 ApplicationKey key = new ApplicationKey(hasher.hash(nationalId), type.get(), YearMonth.from(today));
78 boolean duplicated = record ? !repository.tryRecord(key) : repository.exists(key);
79 if (duplicated) return result(ALREADY_APPLIED_THIS_MONTH, now, firstOpenOfNextMonth(type.get(), today));
80
81 return result(ELIGIBLE, now, null);
82 }
